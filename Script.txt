local function GetHumanoid()

	local char = LP.Character
	if not char then return nil end

	return char:FindFirstChildOfClass("Humanoid")
end

local function GetRoot()

	local char = LP.Character
	if not char then return nil end

	return char:FindFirstChild("HumanoidRootPart")
end

local function SwitchTab(name)

	for tabName,page in pairs(Pages) do

		page.Visible = tabName == name

		if tabName == name then
			Tabs[tabName].BackgroundColor3 =
				Color3.fromRGB(65,65,82)
		else
			Tabs[tabName].BackgroundColor3 =
				Color3.fromRGB(30,30,38)
		end
	end
end

for name,button in pairs(Tabs) do
	button.MouseButton1Click:Connect(function()
		SwitchTab(name)
	end)
end

local function RefreshPlayers()

	for _,child in ipairs(PlayerList:GetChildren()) do

		if child:IsA("TextButton") then
			child:Destroy()
		end
	end

	local query = PlayerSearch.Text:lower()

	for _,player in ipairs(Players:GetPlayers()) do

		if player ~= LP then

			local matches =
				query == ""
				or player.Name:lower():find(query,1,true)
				or player.DisplayName:lower():find(query,1,true)

			if matches then

				local B = MakeButton(
					PlayerList,
					player.DisplayName.."  @"..player.Name
				)

				B.Size = UDim2.new(1,0,0,30)

				B.MouseButton1Click:Connect(function()

					SelectedPlayer = player

					Notify("Selected "..player.DisplayName)
				end)
			end
		end
	end
end

PlayerSearch:GetPropertyChangedSignal("Text"):Connect(RefreshPlayers)

Players.PlayerAdded:Connect(RefreshPlayers)
Players.PlayerRemoving:Connect(function(player)

	if SelectedPlayer == player then
		SelectedPlayer = nil
	end

	RefreshPlayers()
end)

		SelectedPlayer.Character and
		SelectedPlayer.Character:FindFirstChildOfClass("Humanoid")

	if hum and targetHum then

		local success,description =
			pcall(function()
				return targetHum:GetAppliedDescription()

				Notify("Avatar copy failed")
			end
		end
	end

local FireButton = MakeButton(EffectsPage,"Fire Aura: OFF")

local function RemoveFire()

	local char = LP.Character

	if not char then
		return
	end

	for _,obj in ipairs(char:GetDescendants()) do

		if obj:IsA("Fire") and obj.Name == "ShawFire" then
			obj:Destroy()
		end
	end
end

local function AddFire()

	local char = LP.Character

	if not char then
		return
	end

	RemoveFire()

	for _,part in ipairs(char:GetChildren()) do

		if part:IsA("BasePart") then

			local fire = Instance.new("Fire")
			fire.Name = "ShawFire"
			fire.Size = 5
			fire.Heat = 4
			fire.Parent = part
		end
	end
end

FireButton.MouseButton1Click:Connect(function()

	FIRE_AURA = not FIRE_AURA

	FireButton.Text =
		"Fire Aura: "..(FIRE_AURA and "ON" or "OFF")

	if FIRE_AURA then
		AddFire()
	else
		RemoveFire()
	end
end)

RemoveEffects.MouseButton1Click:Connect(function()

	FIRE_AURA = false

	RemoveFire()

	FireButton.Text = "Fire Aura: OFF"

	Notify("Effects removed")
end)

-- FPS
local frames = 0
local lastFPSUpdate = os.clock()

RunService.RenderStepped:Connect(function()

	frames += 1

	if os.clock() - lastFPSUpdate >= 1 then

		FPSLabel.Text = "FPS: "..frames

		frames = 0
		lastFPSUpdate = os.clock()
	end

	local root = GetRoot()

	if root then

		local p = root.Position

		CoordinateLabel.Text =
			string.format(
				"Position: %.0f, %.0f, %.0f",
				p.X,p.Y,p.Z
			)
	end
end)

			end)

		if success then
			PingLabel.Text =
				"Ping: "..math.floor(ping).." ms"
		end
	end
end)

--============================================================--
-- DRAG MAIN
--============================================================--

local dragging = false
local dragStart
local startPos

Header.InputBegan:Connect(function(input)

	if input.UserInputType == Enum.UserInputType.MouseButton1
		or input.UserInputType == Enum.UserInputType.Touch then

		dragging = true
		dragStart = input.Position
		startPos = Main.Position
	end
end)

UIS.InputChanged:Connect(function(input)

	if dragging and (
		input.UserInputType == Enum.UserInputType.MouseMovement
		or input.UserInputType == Enum.UserInputType.Touch
	) then

		local delta = input.Position - dragStart

		Main.Position = UDim2.new(
			startPos.X.Scale,
			startPos.X.Offset + delta.X,
			startPos.Y.Scale,
			startPos.Y.Offset + delta.Y
		)
	end
end)

UIS.InputEnded:Connect(function(input)

	if input.UserInputType == Enum.UserInputType.MouseButton1
		or input.UserInputType == Enum.UserInputType.Touch then

		dragging = false
	end
end)

-- CHARACTER SETUP
--============================================================--

local function SetupCharacter(character)

	local hum = character:WaitForChild("Humanoid",5)

	if hum then

		hum.WalkSpeed =
			SPRINT and SPRINT_SPEED or WALK_SPEED

		hum.UseJumpPower = true
		hum.JumpPower = JUMP_POWER
	end

	table.clear(savedCollision)

	task.wait(.5)

	if FIRE_AURA then
		AddFire()
	end
end

LP.CharacterAdded:Connect(SetupCharacter)

if LP.Character then
	task.spawn(SetupCharacter,LP.Character)
end

-- Search only selects the appropriate tab.
-- It NEVER deletes, hides, or modifies controls.

local SearchMap = {
	["speed"] = "Move",
	["walkspeed"] = "Move",
	["jump"] = "Move",
	["jumppower"] = "Move",
	["noclip"] = "Move",
	["infinite"] = "Move",
	["sprint"] = "Move",

	["player"] = "Players",
	["players"] = "Players",
	["teleport"] = "Players",
	["avatar"] = "Players",
	["skin"] = "Players",
	["spectate"] = "Players",

	["fire"] = "Effects",
	["effect"] = "Effects",
	["fps"] = "Effects",
	["ping"] = "Effects",
	["position"] = "Effects",
	["coordinates"] = "Effects",

	["reset"] = "Settings",
	["settings"] = "Settings",
	["hide"] = "Settings",

	["info"] = "Info",
	["shaw"] = "Info",
	["creator"] = "Info",
	["version"] = "Info"
}

Search.FocusLost:Connect(function()

	local query =
		Search.Text:lower():gsub("%s+","")

	if query == "" then
		return
	end

	for keyword,tab in pairs(SearchMap) do

		if query:find(keyword,1,true)
			or keyword:find(query,1,true) then

			SwitchTab(tab)
			Notify("Found: "..keyword)

			return
		end
	end

	Notify("Nothing found")
end)
